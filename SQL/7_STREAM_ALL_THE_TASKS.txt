USE SCHEMA ANALYTICS;

--Task to refresh churn overview KPI

CREATE OR REPLACE TASK REFRESH_KPI_CHURN_OVERVIEW
WAREHOUSE = COMPUTE_WH
SCHEDULE = 'USING CRON 0 */6 * * * UTC'
WHEN
  SYSTEM$STREAM_HAS_DATA('CLEAN.TELCO_CHURN_STREAM')
AS
CREATE OR REPLACE TABLE KPI_CHURN_OVERVIEW AS
SELECT
    COUNT(*) AS TOTAL_CUSTOMERS,
    SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) AS CHURNED_CUSTOMERS,
    SUM(CASE WHEN CHURN = 'No' THEN 1 ELSE 0 END) AS RETAINED_CUSTOMERS,
    ROUND(
        SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) * 100.0 / COUNT(*),
        2
    ) AS CHURN_RATE_PERCENT
FROM CLEAN.TELCO_CUSTOMER_CHURN;

--Activate the Task

ALTER TASK REFRESH_KPI_CHURN_OVERVIEW RESUME;

SHOW TASKS;

--Manual Run

EXECUTE TASK REFRESH_KPI_CHURN_OVERVIEW;

SELECT * FROM ANALYTICS.KPI_CHURN_OVERVIEW;

--Suspend the ROOT task

ALTER TASK TELCO_DB.ANALYTICS.REFRESH_KPI_CHURN_OVERVIEW SUSPEND;

SHOW TASKS LIKE 'REFRESH_KPI_CHURN_OVERVIEW' 
IN SCHEMA TELCO_DB.ANALYTICS;

--Now DROP tasks

DROP TASK IF EXISTS TELCO_DB.ANALYTICS.REFRESH_KPI_CONTRACT;
DROP TASK IF EXISTS TELCO_DB.ANALYTICS.REFRESH_KPI_CHURN_OVERVIEW;

CREATE TASK REFRESH_KPI_CHURN_OVERVIEW
WAREHOUSE = COMPUTE_WH
SCHEDULE = 'USING CRON 0 */6 * * * UTC'
WHEN SYSTEM$STREAM_HAS_DATA('TELCO_DB.CLEAN.TELCO_CHURN_STREAM')
AS
CREATE OR REPLACE TABLE KPI_CHURN_OVERVIEW AS
SELECT
    COUNT(*) AS TOTAL_CUSTOMERS,
    SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) AS CHURNED_CUSTOMERS,
    SUM(CASE WHEN CHURN = 'No' THEN 1 ELSE 0 END) AS RETAINED_CUSTOMERS,
    ROUND(
        SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) * 100.0 / COUNT(*),
        2
    ) AS CHURN_RATE_PERCENT
FROM TELCO_DB.CLEAN.TELCO_CUSTOMER_CHURN;

--Task Dependency Chain

CREATE TASK REFRESH_KPI_CONTRACT
WAREHOUSE = COMPUTE_WH
AFTER REFRESH_KPI_CHURN_OVERVIEW
AS
CREATE OR REPLACE TABLE KPI_CHURN_BY_CONTRACT AS
SELECT
    CONTRACT,
    COUNT(*) AS TOTAL_CUSTOMERS,
    ROUND(
        SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) * 100.0 / COUNT(*),
        2
    ) AS CHURN_RATE
FROM TELCO_DB.CLEAN.TELCO_CUSTOMER_CHURN
GROUP BY CONTRACT;

ALTER TASK REFRESH_KPI_CHURN_OVERVIEW RESUME;

SHOW TASKS IN SCHEMA TELCO_DB.ANALYTICS;
