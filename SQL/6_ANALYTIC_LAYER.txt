--ANALYTICS LAYER : CUSTOMER CHURN KPIs
USE ROLE ACCOUNTADMIN;
USE DATABASE TELCO_DB;
USE SCHEMA ANALYTICS;


--1.Core KPI – Overall Churn Metrics

CREATE OR REPLACE TABLE ANALYTICS.KPI_CHURN_OVERVIEW AS
SELECT
    COUNT(*) AS TOTAL_CUSTOMERS,
    SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) AS CHURNED_CUSTOMERS,
    SUM(CASE WHEN CHURN = 'No' THEN 1 ELSE 0 END) AS RETAINED_CUSTOMERS,
    ROUND(
        SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) * 100.0 / COUNT(*),
        2
    ) AS CHURN_RATE_PERCENT
FROM CLEAN.TELCO_CUSTOMER_CHURN;  

SELECT * FROM ANALYTICS.KPI_CHURN_OVERVIEW;

--2.Churn by Contract Type (Major Driver)

CREATE OR REPLACE TABLE ANALYTICS.KPI_CHURN_BY_CONTRACT AS
SELECT
    CONTRACT,
    COUNT(*) AS TOTAL_CUSTOMERS,
    SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) AS CHURNED_CUSTOMERS,
    ROUND(
        SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) * 100.0 / COUNT(*),
        2
    ) AS CHURN_RATE_PERCENT
FROM CLEAN.TELCO_CUSTOMER_CHURN
GROUP BY CONTRACT; 

--3.Churn by Internet Service

CREATE OR REPLACE TABLE ANALYTICS.KPI_CHURN_BY_INTERNET AS
SELECT
    INTERNETSERVICE,
    COUNT(*) AS TOTAL_CUSTOMERS,
    SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) AS CHURNED_CUSTOMERS,
    ROUND(
        SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) * 100.0 / COUNT(*),
        2
    ) AS CHURN_RATE_PERCENT
FROM CLEAN.TELCO_CUSTOMER_CHURN
GROUP BY INTERNETSERVICE; 

--4.Churn by Tenure Bucket

CREATE OR REPLACE TABLE ANALYTICS.KPI_CHURN_BY_TENURE AS
SELECT
    CASE
        WHEN TENURE < 12 THEN '0–1 Year'
        WHEN TENURE BETWEEN 12 AND 24 THEN '1–2 Years'
        WHEN TENURE BETWEEN 25 AND 48 THEN '2–4 Years'
        ELSE '4+ Years'
    END AS TENURE_GROUP,
    COUNT(*) AS TOTAL_CUSTOMERS,
    SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) AS CHURNED_CUSTOMERS,
    ROUND(
        SUM(CASE WHEN CHURN = 'Yes' THEN 1 ELSE 0 END) * 100.0 / COUNT(*),
        2
    ) AS CHURN_RATE_PERCENT
FROM CLEAN.TELCO_CUSTOMER_CHURN
GROUP BY TENURE_GROUP
ORDER BY TENURE_GROUP;           

--5.Revenue Impact of Churn

CREATE OR REPLACE TABLE ANALYTICS.KPI_REVENUE_LOSS AS
SELECT
    ROUND(SUM(TOTALCHARGES), 2) AS TOTAL_REVENUE_LOST
FROM CLEAN.TELCO_CUSTOMER_CHURN
WHERE CHURN = 'Yes'; 

--6.High-Risk Customer Segment

CREATE OR REPLACE VIEW ANALYTICS.HIGH_RISK_CUSTOMERS AS
SELECT
    CUSTOMERID,
    TENURE,
    MONTHLYCHARGES,
    CONTRACT,
    INTERNETSERVICE
FROM CLEAN.TELCO_CUSTOMER_CHURN
WHERE CHURN = 'Yes'
AND TENURE < 12
AND MONTHLYCHARGES > 70;
