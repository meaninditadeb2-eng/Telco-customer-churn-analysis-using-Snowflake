USE DATABASE TELCO_DB;
USE SCHEMA ANALYTICS;

CREATE OR REPLACE VIEW ANALYTICS.VW_CUSTOMER_METRICS AS
SELECT
    CUSTOMERID,
    TENURE,
    MONTHLYCHARGES,
    TOTALCHARGES,
    CASE 
        WHEN TENURE < 12 THEN 'NEW'
        WHEN TENURE BETWEEN 12 AND 36 THEN 'MID'
        ELSE 'LONG_TERM'
    END AS CUSTOMER_LIFECYCLE,
    CHURN
FROM CLEAN.TELCO_CUSTOMER_CHURN;
