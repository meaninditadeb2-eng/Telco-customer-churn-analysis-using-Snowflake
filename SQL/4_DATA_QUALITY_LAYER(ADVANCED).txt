USE DATABASE TELCO_DB;
USE SCHEMA ANALYTICS;

CREATE OR REPLACE TABLE ANALYTICS.DATA_QUALITY_REPORT AS
SELECT
    CURRENT_DATE() AS CHECK_DATE,
    COUNT(*) AS TOTAL_RECORDS,
    COUNT_IF(CUSTOMERID IS NULL) AS NULL_CUSTOMER_ID,
    COUNT_IF(TOTALCHARGES IS NULL) AS NULL_TOTAL_CHARGES,
    COUNT_IF(TENURE < 0) AS INVALID_TENURE
FROM CLEAN.TELCO_CUSTOMER_CHURN;
