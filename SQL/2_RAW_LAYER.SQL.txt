-- Set context
USE ROLE ACCOUNTADMIN;
USE DATABASE TELCO_DB;
USE SCHEMA RAW;

-- Create file format
CREATE OR REPLACE FILE FORMAT TELCO_CSV_FORMAT
TYPE = CSV
FIELD_DELIMITER = ','
SKIP_HEADER = 1
FIELD_OPTIONALLY_ENCLOSED_BY = '"'
NULL_IF = ('NULL','null','');

-- Create raw table
CREATE OR REPLACE TABLE TELCO_CUSTOMER_CHURN (
    CUSTOMERID VARCHAR(50),
    GENDER VARCHAR(10),
    SENIORCITIZEN NUMBER(1,0),
    PARTNER VARCHAR(5),
    DEPENDENTS VARCHAR(5),
    TENURE NUMBER(3,0),
    PHONESERVICE VARCHAR(5),
    MULTIPLELINES VARCHAR(20),
    INTERNETSERVICE VARCHAR(20),
    ONLINESECURITY VARCHAR(20),
    ONLINEBACKUP VARCHAR(20),
    DEVICEPROTECTION VARCHAR(20),
    TECHSUPPORT VARCHAR(20),
    STREAMINGTV VARCHAR(20),
    STREAMINGMOVIES VARCHAR(20),
    CONTRACT VARCHAR(20),
    PAPERLESSBILLING VARCHAR(5),
    PAYMENTMETHOD VARCHAR(30),
    MONTHLYCHARGES NUMBER(8,2),
    TOTALCHARGES VARCHAR(20),
    CHURN VARCHAR(5)
);
-- Load data
COPY INTO TELCO_CUSTOMER_CHURN
FROM @TELCO_STAGE
FILE_FORMAT = TELCO_CSV_FORMAT
ON_ERROR = 'CONTINUE';

TRUNCATE TABLE RAW.TELCO_CUSTOMER_CHURN;

COPY INTO RAW.TELCO_CUSTOMER_CHURN
FROM @TELCO_STAGE
FILE_FORMAT = (FORMAT_NAME = TELCO_CSV_FORMAT)
ON_ERROR = 'CONTINUE';

--Row count checking
SELECT COUNT(*) AS RAW_ROW_COUNT
FROM RAW.TELCO_CUSTOMER_CHURN;
