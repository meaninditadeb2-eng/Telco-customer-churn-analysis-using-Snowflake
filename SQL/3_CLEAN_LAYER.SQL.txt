-- Set context
USE ROLE ACCOUNTADMIN;
USE DATABASE TELCO_DB;
USE SCHEMA CLEAN;

CREATE OR REPLACE TABLE TELCO_CUSTOMER_CHURN AS
SELECT
    CUSTOMERID,
    UPPER(GENDER) AS GENDER,
    SENIORCITIZEN,
    PARTNER,
    DEPENDENTS,
    TENURE,
    PHONESERVICE,
    MULTIPLELINES,
    INTERNETSERVICE,
    ONLINESECURITY,
    ONLINEBACKUP,
    DEVICEPROTECTION,
    TECHSUPPORT,
    STREAMINGTV,
    STREAMINGMOVIES,
    CONTRACT,
    PAPERLESSBILLING,
    PAYMENTMETHOD,
    MONTHLYCHARGES,
    TRY_TO_NUMBER(TOTALCHARGES) AS TOTALCHARGES,
    CHURN
FROM RAW.TELCO_CUSTOMER_CHURN;

SELECT COUNT(*) AS RAW_ROW_COUNT
FROM TELCO_CUSTOMER_CHURN;
